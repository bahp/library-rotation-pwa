Scan / Search → CatalogService (Parallel Providers) → Normalizer + Merge → RuleTagger → Store (IndexedDB) → patchRootHtml()
  • No cover flicker: src/core/dom-patch.js diffs live DOM nodes in place so unchanged <img> book covers never unmount or re-fetch on navigation.

3. Add a Custom MetadataProvider in 2 Steps

Every catalog source implements MetadataProvider (lookupByIsbn + searchBooks) and returns normalized BookCandidate objects. CatalogService runs all providers via Promise.allSettled, merges ISBN matches, and stamps provenance (sources).

1

Create Provider Module

Add src/features/catalog/providers/my-provider.js implementing lookupByIsbn and searchBooks.

2

Register in CatalogService

Add it to the default array in createCatalogService() inside src/features/catalog/catalog-service.js.

// src/features/catalog/providers/national-library-provider.js
import { normalizeIsbn } from '../isbn.js';

/**
 * @returns {import('../catalog-service.js').MetadataProvider}
 */
export function createNationalLibraryProvider({ fetchImpl = globalThis.fetch } = {}) {
  return {
    name: 'national-library',

    async lookupByIsbn(rawIsbn) {
      const isbn = normalizeIsbn(rawIsbn);
      if (!isbn) return [];
      const res = await fetchImpl(`https://example-library.org/api/isbn/${isbn}`);
      if (!res.ok) return [];
      const item = await res.json();
      return [
        {
          isbn,
          title: item.title,
          authors: item.authors || ['Unknown Author'],
          coverUrl: item.cover || '',
          coverCandidates: item.cover ? [item.cover] : [],
          publisher: item.publisher || '',
          publishedDate: item.year || '',
          pageCount: Number(item.pages) || 32,
          description: item.summary || '',
          categories: item.subjects || [],
          language: item.lang || 'en',
          format: 'paperback',
          source: 'manual',
          sources: ['manual'],
        },
      ];
    },

    async searchBooks({ title, author }) {
      const q = [title, author].filter(Boolean).join(' ');
      if (!q) return [];
      // Query search endpoint and return BookCandidate[]
      return [];
    },
  };
}
